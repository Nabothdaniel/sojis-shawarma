rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // ==========================================
    // USERS COLLECTION
    // ==========================================
    match /users/{userId} {
      // Users can only read and write their own documents. 
      // Admins (role == 'admin') can read all users (e.g. for dashboard)
      allow read: if request.auth != null && (request.auth.uid == userId || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin');
      
      // Users can update their own document (e.g. favorite_products array)
      allow update: if request.auth != null && request.auth.uid == userId;
      
      // Only admins can delete user documents entirely, or users themselves if account deletion is allowed
      allow delete: if request.auth != null && request.auth.uid == userId;
    }

    // ==========================================
    // PRODUCTS COLLECTION
    // ==========================================
    match /products/{productId} {
      // Anyone can view products (guests included, because the catalog is public)
      allow read: if true;
      
      // Only admins can add, update, or delete products
      allow create, update, delete: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // ==========================================
    // CATEGORIES COLLECTION
    // ==========================================
    match /categories/{categoryId} {
      // Anyone can view categories
      allow read: if true;
      
      // Only admins can mutate categories
      allow create, update, delete: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // ==========================================
    // ORDERS COLLECTION
    // ==========================================
    match /orders/{orderId} {
      // Users can only read their own orders. Note that anonymous guests can create orders, 
      // but only identified users or admins can read histories.
      allow read: if request.auth != null && (
        resource.data.user_id == request.auth.uid || 
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'
      );
      
      // Anyone can CREATE an order (guests included, provided they submit required data)
      allow create: if true;
      
      // Only admins can UPDATE an order (change status, confirm payment) 
      // OR users can update their order IF they are just uploading a receipt (payment_status update)
      // *This can be further restricted by field!*
      allow update: if request.auth != null && (
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin' ||
        (resource.data.user_id == request.auth.uid)
      );
      
      allow delete: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // ==========================================
    // SETTINGS COLLECTION
    // ==========================================
    match /settings/{settingId} {
      // Everyone can read store settings (like payment details)
      allow read: if true;
      
      // Only admins can change store settings
      allow write: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }
  }
}
